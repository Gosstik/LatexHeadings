%%% Usage: \begin{vscodecode}{file_name.cpp}{C++} ... \end{vscodecode}
%%% The filename may be empty. Languages come from listings, plus yaml above.
\newtcblisting{vscodecode}[2]{
    enhanced,
    breakable,
    listing only,
    listing engine=listings,
    colback=VSCodeLightBackground,
    colframe=VSCodeLightFrame,
    colbacktitle=VSCodeLightTitle,
    coltitle=VSCodeLightText,
    boxrule=0.45pt,
    boxsep=4pt,
    arc=1.8mm,
    outer arc=1.8mm,
    left=5.0mm,
    right=1.5mm,
    top=0pt,
    bottom=0pt,
    toptitle=1.1mm,
    bottomtitle=1.1mm,
    before skip=2mm,
    after skip=3mm,
    fonttitle=\ttfamily\footnotesize,
    title={%
        \detokenize{#1}
        \hfill
        #2%
    },
    % Keep this order: loading the language first prevents it from
    % discarding the custom fade delimiter from the style.
    listing options={
        language=#2,
        style=vscode-light,
    },
    overlay={
        \draw[
            VSCodeLightFrame,
            line width=0.35pt
        ]
        ([xshift=6.1mm]interior.south west)
        --
        ([xshift=6.1mm]interior.north west);
    },
}
